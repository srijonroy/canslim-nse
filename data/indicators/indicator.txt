//@version=5
// ════════════════════════════════════════════════════════════════════════════
//  Brooks Regime & Always-In
//  A price-action assistant inspired by Al Brooks' framework.
//
//  It answers two questions on every bar:
//    1) PROBABILITY that the market is TRENDING vs. in a TRADING RANGE
//    2) The current ALWAYS-IN direction (Long / Short / Neutral)
//
//  NOTE: The probability is an evidence-weighted heuristic (not a trained ML
//  model — Pine can't train one). It combines several objective, tunable
//  measures that each vote "trend" or "range", then blends them 0–100%.
// ════════════════════════════════════════════════════════════════════════════
indicator("Brooks Regime & Always-In", shorttitle="AB Regime/AI", overlay=true, max_labels_count=500)

// ─── Core inputs ───────────────────────────────────────────────────────────────
grpC   = "Core"
emaLen = input.int(20, "EMA Length",  minval=1, group=grpC)
atrLen = input.int(14, "ATR Length",  minval=1, group=grpC)

// ─── Trend-vs-Range probability inputs ─────────────────────────────────────────
grpP        = "Trend vs Range Probability"
erLen       = input.int(14,  "Efficiency Ratio Lookback", minval=2,  group=grpP)
adxLen      = input.int(14,  "ADX / DMI Length",          minval=1,  group=grpP)
slopeLen    = input.int(5,   "EMA Slope Lookback",        minval=1,  group=grpP)
slopeSens   = input.float(0.30, "EMA Slope Full-Trend (xATR/bar)", step=0.05, minval=0.05, group=grpP)
sideLen     = input.int(20,  "EMA Side / Cross Lookback",  minval=2,  group=grpP)
smoothLen   = input.int(3,   "Probability Smoothing",      minval=1,  group=grpP)
trendThresh = input.float(55, "Trend Threshold %", minval=50, maxval=95, group=grpP)

// Evidence weights (need not sum to 1 — they are normalized)
wER    = input.float(0.30, "Weight · Efficiency Ratio", minval=0, group=grpP)
wADX   = input.float(0.25, "Weight · ADX",              minval=0, group=grpP)
wSlope = input.float(0.20, "Weight · EMA Slope",        minval=0, group=grpP)
wSide  = input.float(0.15, "Weight · EMA Side Bias",    minval=0, group=grpP)
wCross = input.float(0.10, "Weight · EMA Cross (inv)",  minval=0, group=grpP)

// ─── Always-In inputs ──────────────────────────────────────────────────────────
grpAI      = "Always In"
bodyFactor = input.float(0.50, "Trend-Bar Body ≥ xATR", step=0.05, minval=0.05, group=grpAI)
showFlipLabels = input.bool(true,  "Show AIL/AIS Flip Labels on Candles", group=grpAI)
colorEma   = input.bool(true,  "Color EMA by Always-In",            group=grpAI)
flipLabelSize = input.string("Small", "Flip Label Size", options=["Tiny","Small","Normal"], group=grpAI)

// ─── Core calculations ─────────────────────────────────────────────────────────
ema = ta.ema(close, emaLen)
atr = ta.atr(atrLen)

// 1) Efficiency Ratio (Kaufman): net move / total path. 1 = perfectly directional.
change = math.abs(close - close[erLen])
vol    = math.sum(math.abs(close - close[1]), erLen)
er     = vol == 0 ? 0.0 : change / vol

// 2) ADX normalized (0..1). ADX ~40+ = very strong trend.
[diP, diM, adxVal] = ta.dmi(adxLen, adxLen)
adxN = math.min(nz(adxVal) / 40.0, 1.0)

// 3) EMA slope per bar, normalized by ATR (0..1).
slopePerBar = (ema - ema[slopeLen]) / slopeLen
slopeN      = atr == 0 ? 0.0 : math.min(math.abs(slopePerBar) / (atr * slopeSens), 1.0)

// 4) EMA side bias: how one-sided are the recent closes around the EMA (0..1).
aboveFrac = math.sum(close > ema ? 1 : 0, sideLen) / sideLen
sideBias  = math.abs(aboveFrac - 0.5) * 2

// 5) EMA cross frequency (a range symptom) → inverted so more crosses = less trend.
crossCount = math.sum(ta.cross(close, ema) ? 1 : 0, sideLen)
crossN     = math.min(crossCount / (sideLen / 2.0), 1.0)
trendFromCross = 1.0 - crossN

// ─── Blend the evidence into a Trend probability ───────────────────────────────
wSum = wER + wADX + wSlope + wSide + wCross
trendScore = wSum == 0 ? 0.0 :
     (nz(er)*wER + adxN*wADX + nz(slopeN)*wSlope + nz(sideBias)*wSide + nz(trendFromCross)*wCross) / wSum

trendProb = ta.ema(nz(trendScore) * 100.0, smoothLen)
rangeProb = 100.0 - trendProb

isTrend   = trendProb >= trendThresh
regimeTxt = isTrend ? "TREND" : "TRADING RANGE"

// ─── Always-In state machine ───────────────────────────────────────────────────
// Flip LONG on a strong bull trend bar that closes above the EMA and breaks the
// prior bar high; flip SHORT on the mirror image; otherwise hold the last state.
body       = math.abs(close - open)
strongBull = body >= atr * bodyFactor and close > ema and close > high[1]
strongBear = body >= atr * bodyFactor and close < ema and close < low[1]

var int ai = 0
if strongBull
    ai := 1
else if strongBear
    ai := -1

aiTxt   = ai == 1 ? "LONG" : ai == -1 ? "SHORT" : "NEUTRAL"
aiColor = ai == 1 ? color.green : ai == -1 ? color.red : color.gray

// ─── Always-In flip labels (replaces background coloring) ─────────────────────
flipToLong  = ai == 1  and ai[1] != 1
flipToShort = ai == -1 and ai[1] != -1

flipLblSize = flipLabelSize == "Tiny"  ? size.tiny  :
              flipLabelSize == "Normal" ? size.normal : size.small

if showFlipLabels and flipToLong
    lblL = label.new(bar_index, low, text="AIL", style=label.style_label_up, yloc=yloc.belowbar, color=color.new(color.green, 0), textcolor=color.white, size=flipLblSize)
if showFlipLabels and flipToShort
    lblS = label.new(bar_index, high, text="AIS", style=label.style_label_down, yloc=yloc.abovebar, color=color.new(color.red, 0), textcolor=color.white, size=flipLblSize)

// ─── Visuals ───────────────────────────────────────────────────────────────────
emaPlotColor = colorEma ? aiColor : color.blue
plot(ema, "EMA 20", color=color.new(emaPlotColor, 0), linewidth=2)

// ─── Dashboard table ───────────────────────────────────────────────────────────
probColor  = color.from_gradient(trendProb, 0, 100, color.new(color.red, 0), color.new(color.green, 0))
regimeCol  = isTrend ? color.new(color.green, 20) : color.new(color.orange, 20)

var table dash = table.new(position.top_right, 2, 5, border_width=1, border_color=color.new(color.gray, 50))
if barstate.islast
    table.cell(dash, 0, 0, "Regime",     text_color=color.white, bgcolor=color.new(color.black, 20), text_size=size.normal)
    table.cell(dash, 1, 0, regimeTxt,    text_color=color.white, bgcolor=regimeCol,                  text_size=size.normal)
    table.cell(dash, 0, 1, "Trend %",    text_color=color.white, bgcolor=color.new(color.black, 20))
    table.cell(dash, 1, 1, str.tostring(trendProb, "#.0") + "%", text_color=color.white, bgcolor=probColor)
    table.cell(dash, 0, 2, "Range %",    text_color=color.white, bgcolor=color.new(color.black, 20))
    table.cell(dash, 1, 2, str.tostring(rangeProb, "#.0") + "%", text_color=color.white, bgcolor=color.new(color.gray, 20))
    table.cell(dash, 0, 3, "Always In",  text_color=color.white, bgcolor=color.new(color.black, 20), text_size=size.normal)
    table.cell(dash, 1, 3, aiTxt,        text_color=color.white, bgcolor=aiColor,                    text_size=size.normal)
    table.cell(dash, 0, 4, "ADX",        text_color=color.white, bgcolor=color.new(color.black, 20))
    table.cell(dash, 1, 4, str.tostring(nz(adxVal), "#.0"),      text_color=color.white, bgcolor=color.new(color.gray, 20))

// ─── Alerts ────────────────────────────────────────────────────────────────────
alertcondition(ai == 1  and ai[1] != 1,  "Always-In → LONG",  "Always-In flipped LONG")
alertcondition(ai == -1 and ai[1] != -1, "Always-In → SHORT", "Always-In flipped SHORT")
alertcondition(isTrend and not isTrend[1], "Regime → TREND",         "Regime changed to TREND")
alertcondition(not isTrend and isTrend[1], "Regime → TRADING RANGE", "Regime changed to TRADING RANGE")



//---------------------------------------------------------------------


// ─── EMA 20 for each timeframe ────────────────────────────────────────────────
grpMTF = "Multi-Timeframe EMAs"
show5m  = input.bool(true, "Show 5m EMA",  group=grpMTF, inline="5m")
col5m   = input.color(color.blue,    "", group=grpMTF, inline="5m")
lw5m    = input.int(1, "Width", minval=1, maxval=10, group=grpMTF, inline="5m")

show15m = input.bool(true, "Show 15m EMA", group=grpMTF, inline="15m")
col15m  = input.color(color.orange,  "", group=grpMTF, inline="15m")
lw15m   = input.int(2, "Width", minval=1, maxval=10, group=grpMTF, inline="15m")

show60m = input.bool(true, "Show 60m EMA", group=grpMTF, inline="60m")
col60m  = input.color(color.red,     "", group=grpMTF, inline="60m")
lw60m   = input.int(3, "Width", minval=1, maxval=10, group=grpMTF, inline="60m")

show4h  = input.bool(true, "Show 4H EMA",  group=grpMTF, inline="4h")
col4h   = input.color(color.purple,  "", group=grpMTF, inline="4h")
lw4h    = input.int(3, "Width", minval=1, maxval=10, group=grpMTF, inline="4h")

show1d  = input.bool(true, "Show 1D EMA",  group=grpMTF, inline="1d")
col1d   = input.color(color.fuchsia, "", group=grpMTF, inline="1d")
lw1d    = input.int(4, "Width", minval=1, maxval=10, group=grpMTF, inline="1d")

ema20_5m  = request.security(syminfo.tickerid, "5",   ta.ema(close, 20), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_on)
ema20_15m = request.security(syminfo.tickerid, "15",  ta.ema(close, 20), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_on)
ema20_60m = request.security(syminfo.tickerid, "60",  ta.ema(close, 20), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_on)
ema20_4h  = request.security(syminfo.tickerid, "240", ta.ema(close, 20), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_on)
ema20_1d  = request.security(syminfo.tickerid, "D",   ta.ema(close, 20), gaps=barmerge.gaps_off, lookahead=barmerge.lookahead_on)

// ─── Plots ────────────────────────────────────────────────────────────────────
plot(show5m  ? ema20_5m  : na, title="EMA 20 (5m)",  color=color.new(col5m,  0), linewidth=lw5m)
plot(show15m ? ema20_15m : na, title="EMA 20 (15m)", color=color.new(col15m, 0), linewidth=lw15m)
plot(show60m ? ema20_60m : na, title="EMA 20 (60m)", color=color.new(col60m, 0), linewidth=lw60m)
plot(show4h  ? ema20_4h  : na, title="EMA 20 (4H)",  color=color.new(col4h,  0), linewidth=lw4h)
plot(show1d  ? ema20_1d  : na, title="EMA 20 (1D)",  color=color.new(col1d,  0), linewidth=lw1d)

// ─── Bar Counter ──────────────────────────────────────────────────────────────
sizeOption = input.string("Normal", "Label Size", options=["Auto","Huge","Large","Normal","Small","Tiny"])
labelSize  = sizeOption == "Huge"  ? size.huge  :
             sizeOption == "Large" ? size.large :
             sizeOption == "Small" ? size.small :
             sizeOption == "Tiny"  ? size.tiny  :
             sizeOption == "Auto"  ? size.auto  : size.normal

c_labelColor = input.color(color.orange, "Text Color")
c_contador   = input.int(2, "Display at every X bars")

is_new_day() =>
    d = dayofweek
    na(d[1]) or d != d[1]

var count = 1

if is_new_day()
    count := 1
else
    count := count + 1

if count % c_contador == 0
    label1 = label.new(bar_index, 0, style=label.style_none, text=str.tostring(count))
    label.set_textcolor(label1, c_labelColor)
    label.set_yloc(label1, yloc.belowbar)
    label.set_size(label1, labelSize)

// ─── Inside / Outside Bar Patterns ─────────────────────────────────────────────
// An inside bar:  high lower than prior high  AND low higher than prior low
// An outside bar: high higher than prior high AND low lower than prior low
showPatterns = input.bool(true, "Show Inside/Outside Patterns")
c_patternColor = input.color(color.yellow, "Pattern Text Color")
patternSizeOption = input.string("Huge", "Pattern Label Size", options=["Auto","Huge","Large","Normal","Small","Tiny"])
patternSize = patternSizeOption == "Huge"  ? size.huge  :
              patternSizeOption == "Large" ? size.large :
              patternSizeOption == "Small" ? size.small :
              patternSizeOption == "Tiny"  ? size.tiny  :
              patternSizeOption == "Auto"  ? size.auto  : size.normal

isInside  = high < high[1] and low > low[1]
isOutside = high > high[1] and low < low[1]

// Patterns evaluated on the current bar looking back
pat_iii = isInside and isInside[1] and isInside[2]        // inside inside inside
pat_ii  = isInside and isInside[1]                        // inside inside
pat_ioi = isInside and isOutside[1] and isInside[2]        // inside outside inside

// NEW: Outside bar patterns (mirror of the inside bar logic)
pat_ooo = isOutside and isOutside[1] and isOutside[2]      // outside outside outside
pat_oo  = isOutside and isOutside[1]                       // outside outside

// Priority order (avoid duplicate labels on the same bar):
// iii > ooo > ioi > ii > oo
patternText = pat_iii ? "iii" :
              pat_ooo ? "ooo" :
              pat_ioi ? "ioi" :
              pat_ii  ? "ii"  :
              pat_oo  ? "oo"  : na

if showPatterns and not na(patternText)
    label2 = label.new(bar_index, high, style=label.style_label_down, text=patternText)
    label.set_textcolor(label2, c_patternColor)
    label.set_color(label2, color.new(color.blue, 0))
    label.set_yloc(label2, yloc.abovebar)
    label.set_size(label2, patternSize)
